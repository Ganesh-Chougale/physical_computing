## 1. Save this file at PicoW as `myServer.py` 
```python
import socket as piPicoWsock
import time
import network

wifiObj = network.WLAN(network.STA_IF) # creating wifi object
wifiObj.active(True)

# wifiObj.connect("YOUR_WIFI_NAME", "YOUR_WIFI_PASSWORD")
# Example: wifiObj.connect("HomeNetwork", "mh09_us")
wifiObj.connect('CE5', '12345678')

while wifiObj.isconnected() == False:
    print('waiting for connection......')
    time.sleep(1)
#print('connected')
wifiInfo = wifiObj.ifconfig()
serverIpAddress = wifiInfo[0] # first element = acutal IP
print(serverIpAddress)

serverListenerPort = 2222
bufferSize = 1024

UPDserver = piPicoWsock.socket(piPicoWsock.AF_INET, piPicoWsock.SOCK_DGRAM)
UPDserver.bind((serverIpAddress, serverListenerPort))
print('UDP Server Up & waiting....')

while True:
    message,address = UPDserver.recvfrom(bufferSize)
    messageDecoded = message.decode('utf-8')
    print(messageDecoded)
```  
run this programm & grab that Ip address, we need it to assign in next file.  

## 2. Save this second file at local Computer as `myClient.py`
```python
import socket as localSocket

serverIpAddress = '10.161.66.250' # found in first program terminal
serverListenerPort = 2222
serverConfig = (serverIpAddress, serverListenerPort)

bufferSize = 1024

UDPclient = localSocket.socket(localSocket.AF_INET, localSocket.SOCK_DGRAM)

while True:
    userInput = input('What is your command?: ')
    cmdEncoded = userInput.encode('utf-8')
    UDPclient.sendto(cmdEncoded, serverConfig)
```  

Note: to work this both local machine & pico w should be connected on same wifi network
